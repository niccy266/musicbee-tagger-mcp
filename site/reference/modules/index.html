    def cmd_revert_one(applier, stage_id: str) -> str:
    """Revert an applied staged change by restoring its .bak backup."""
    try:
        manifest = applier.load(stage_id)
    except (FileNotFoundError, ValueError):
        logger.error("Staged manifest %s is missing; cannot revert.", stage_id)
        print(f"\n[ERROR] Stage {stage_id} is missing.")
        return "missing_stage"
    if not manifest.get("applied"):
        status = []
        if manifest.get("reverted"):
            status.append("already reverted")
        if manifest.get("withdrawn"):
            status.append("withdrawn")
        if manifest.get("failed"):
            status.append("failed")
        if manifest.get("archived"):
            status.append("archived")
        print(f"\n[SKIP] {stage_id} is {', '.join(status) or 'not applied'}; cannot revert.")
        return "not_applied"
    backup_path = manifest.get("backup_path")
    if not backup_path:
        print(f"\n[SKIP] {stage_id} has no backup path stored; cannot revert.")
        return "no_backup"
    backup = Path(backup_path)
    if not backup.is_file():
        print("\n[ERROR] Backup file not found; cannot revert.")
        return "no_backup_file"

    display_name = manifest.get("display_name", stage_id)
    print()
    print(f"Revert preview for {stage_id}:")
    print(f"  file: {display_name}")
    print(f"  backup: {backup_path}")
    print()
    print("This will restore the original .bak backup over the current file,")
    print("undoing all tag changes that were applied.")
    answer = input(
        "\nEnter=revert/restore backup, q=quit [Enter/q]: "
    ).strip().lower()
    if answer == "q":
        print("Quitting revert.")
        return "quit"
    if answer not in {"", "y", "yes"}:
        print("Unrecognized choice; revert aborted.")
        return "pending"

    try:
        result = applier.revert(stage_id)
    except FileNotFoundError as exc:
        logger.error("Revert failed for %s: %s", stage_id, exc)
        print("\n[ERROR] Revert failed: file not found.")
        return "error"
    if result.get("status") == "already_reverted":
        print(f"\n[SKIP] {stage_id} has already been reverted.")
        return "already_reverted"
    print(f"\n✓ Reverted {stage_id}: {display_name}")
    print(f"  Restored from backup: {backup_path}")
    return "reverted"

    def run(
    input_stream: TextIO = sys.stdin,
    output_stream: TextIO = sys.stdout,
    playlist_path: str = "",
) -> int:
    """Register a supplied playlist or prompt locally, then print path-free details."""
    print(
        "This local command registers playlist entries without sending paths to "
        "the MCP agent.",
        file=output_stream,
    )
    print(
        "Run this command in a separate terminal, not through the agent.",
        file=output_stream,
    )
    if not playlist_path:
        print(
            "Playlist path (.m3u or .m3u8): ",
            end="",
            file=output_stream,
            flush=True,
        )
        playlist_path = input_stream.readline().strip().strip('"')
    if not playlist_path:
        print("No playlist path was entered.", file=output_stream)
        return 1

    playlist_name = playlist_path.replace("\\", "/").rsplit("/", 1)[-1]
    job_id = str(uuid.uuid4())
    get_job_store().create_job([], playlist_name=playlist_name, skipped=0, job_id=job_id)
    print(
        f"Job id: {job_id} (registration in progress)",
        file=output_stream,
        flush=True,
    )

    result = register_playlist(
        playlist_path,
        get_registry(),
        get_job_store(),
        job_id,
        progress_callback=lambda current, total: _progress_bar(
            output_stream, current, total
        ),
    )
    if "error" in result:
        print(f"Unable to register playlist: {result['error']}.", file=output_stream)
        return 1

    registered_count = len(result["registered"])
    print(
        f"Registered {registered_count} audio file(s) as playlist job "
        f"{job_id}.",
        file=output_stream,
    )
    duplicates = result.get("duplicates", [])
    if duplicates:
        count = len(duplicates)
        label = "entry" if count == 1 else "entries"
        print(
            f"Skipped {count} duplicate playlist {label} already registered "
            f"in another active job:",
            file=output_stream,
        )
        for entry in duplicates:
            print(
                f"- {entry.get('display_name')}: existing job "
                f"{entry.get('existing_job_id')}",
                file=output_stream,
            )
    print(
        "Pass this job id to the agent loop or ask the agent for the next "
        "batch; the playlist path was never sent over MCP.",
        file=output_stream,
    )

    skipped = result["skipped"]
    if skipped:
        count = len(skipped)
        label = "entry" if count == 1 else "entries"
        print(f"Skipped {count} playlist {label}:", file=output_stream)
        for entry in skipped:
            print(f"- Line {entry['line']}: {entry['error']}", file=output_stream)

    try:
        groups = get_job_store().album_groups(job_id)
        group_count = groups.get("group_count", 0) if isinstance(groups, dict) else 0
        if group_count:
            print(f"Album groups: {group_count}", file=output_stream)
            for group in groups.get("album_groups", []):
                print(
                    f"- {group.get('album')}: {group.get('track_count', 0)} track(s)",
                    file=output_stream,
                )
        single_count = groups.get("single_count", 0) if isinstance(groups, dict) else 0
        if single_count:
            print(f"Single-track items: {single_count}", file=output_stream)
    except Exception:
        pass

    return 0

    class AuditReport(list):
    """Audit report that is primarily a list of flagged stages.

    Mapping-style access is delegated to the full report dictionary, so callers
    that expect either a list or a dict can use the same object.
    """

    def __init__(self, flagged, report):
        super().__init__(flagged)
        self._report = dict(report)

    def _mapping(self):
        return self._report

    def __getitem__(self, key):
        if isinstance(key, str):
            return self._report[key]
        return super().__getitem__(key)

    def __contains__(self, key):
        if isinstance(key, str):
            return key in self._report
        return super().__contains__(key)

    def get(self, key, default=None):
        if isinstance(key, str):
            return self._report.get(key, default)
        return default

    def keys(self):
        return self._report.keys()

    def values(self):
        return self._report.values()

    def items(self):
        return self._report.items()

    def __repr__(self):
        return repr(self._report)

    class StagingManager:
    """Manages pending tag-change manifests on disk."""

    def __init__(self, staging_dir: Optional[Path] = None) -> None:
        self.store = StageManifestStore(staging_dir)
        self.staging_dir = self.store.directory

    def stage(
        self,
        absolute_path: Path,
        display_name: str,
        proposed_tags: Dict[str, Any],
        mb_recording_id: Optional[str] = None,
        mb_release_id: Optional[str] = None,
        forensic_score: Optional[float] = None,
        match_source: Optional[str] = None,
        match_confidence: Optional[float] = None,
        notes: str = "",
        token: Optional[str] = None,
        job_id: Optional[str] = None,
        batch: Optional[str] = None,
        note: Optional[str] = None,
        rationale: Optional[str] = None,
    ) -> dict:
        """Write a pending change manifest and return a stage_id + summary.

        The absolute_path is stored inside the manifest file on disk.
        It is NEVER returned to the caller.

        Returns:
            {"stage_id": str, "display_name": str, "diff_summary": list, "notes": str}
        """
        # Generate the field-by-field diff (reads file for current values).
        diff = _changed_fields(str(absolute_path), proposed_tags)
        try:
            baseline_tags = read_audio_metadata(str(absolute_path)).get("tags", {})
        except Exception:
            baseline_tags = None

        override_text = (
            manual_override_justification
            or override_justification
            or override_reason
            or justification
        )
        is_manual_override = bool(manual_override or override_text)
        if is_manual_override:
            if not isinstance(override_text, str) or not override_text.strip():
                raise ValueError("justification is required for a manual override")
            override_text = override_text.strip()
        else:
            override_text = None

        if not notes and note:
            notes = note
        if rationale:
            notes = (str(notes).strip() + ", " + str(rationale).strip()).strip() if notes else str(rationale).strip()
        notes = notes if isinstance(notes, str) else ("" if notes is None else str(notes))

        stage_id = str(uuid.uuid4())
        manifest = {
            "stage_id": stage_id,
            "created_at": datetime.now(timezone.utc).isoformat(),
            "previous_tags": baseline_tags,
            "display_name": display_name,
            "absolute_path": str(absolute_path),  # stored on disk, never returned
            "token": token,
            "job_id": job_id,
            "batch": batch,
            "proposed_tags": proposed_tags,
            "diff": diff,
            "mb_recording_id": mb_recording_id,
            "mb_release_id": mb_release_id,
            "forensic_score": forensic_score,
            "match_source": match_source,
            "match_confidence": match_confidence,
            "notes": notes,
            "note": notes,
            "rationale": notes,
            "applied": False,
            "withdrawn": False,
            "revision": 1,
            "history": [],
        }

        if is_manual_override:
            manifest.setdefault("history", []).append({
                "action": "manual_override",
                "at": manifest["created_at"],
                "reason": override_text,
            })

        self.store.save(manifest)

        # Return only safe, non-path fields to the caller
        return {
            "stage_id": stage_id,
            "display_name": display_name,
            "revision": 1,
            "diff_summary": diff,
            "notes": notes,
            "message": (
                f"Staged {len(diff)} field change(s) for '{display_name}'. "
                f"Run `python -m musicbee_tagger_mcp.apply_staged --stage-id {stage_id}` to apply."
            ),
        }

    def stage_manual_override(self, *args, **kwargs):
        kwargs.setdefault("manual_override", True)
        return self.stage(*args, **kwargs)

    def list_pending(self) -> List[dict]:
        """List all pending (not yet applied) staged changes — safe fields only."""
        return self.store.list_pending()
    def list_staged_changes(self, *args, **kwargs):

            "action": "withdrawn",
            "revision": manifest["revision"],
            "at": now,
            "reason": reason,
        })

    manifest = self.store.mutate(stage_id, update)
    return {
        "stage_id": manifest["stage_id"],
        "display_name": manifest["display_name"],
        "status": self._status(manifest),
        "revision": manifest["revision"],

    class FileRegistry:
    """Token-to-path store with optional persistence for local registration.

    Persistent registries let a human-run local command register files before
    the MCP client starts, without sending paths in MCP tool calls.
    """

    def __init__(self, directory: Optional[Path] = None) -> None:
        self.directory = directory
        self._store: Dict[str, Path] = {}  # token → absolute path

    def register(self, path: str, store_path: Optional[str] = None) -> dict:
        """Register a file path and return a token + display name.

        Verifies existence of `path` on disk, but optionally persists `store_path`
        (e.g., container virtual path).

        Returns:
            {"token": str, "display_name": str}
            display_name is the bare filename only — never a path.

        Raises:
            FileNotFoundError: if the path does not exist on disk.
        """
        p = Path(str(path)).expanduser().resolve()
        if not p.is_file():
            raise FileNotFoundError("File not found")
        s = Path(str(store_path)) if store_path else p
        existing = self.token_for_path(s)
        if existing is not None:
            return {"token": existing, "display_name": p.name,
                    "already_registered": True}
        token = str(uuid.uuid4())
        self._store[token] = s
        if self.directory is not None:
            self._save(token, s)
        return {"token": token, "display_name": p.name}

    def register_many(self, paths: list[str]) -> list[dict]:
        """Register a list of paths. Returns list of {token, display_name}."""
        return [self.register(p) for p in paths]

    def resolve(self, token: str) -> Path:
        """Resolve a token to its absolute path (internal use only).

        Raises:
            KeyError: if the token is not registered.
        """
        try:
            normalized_token = str(uuid.UUID(token))
        except (TypeError, ValueError, AttributeError):
            # Support the short aliases shown to agents when they uniquely
            # identify one registered token.  This keeps read_track_metadata and
            # related tools usable if a caller passes an abbreviated token.
            candidates = self._matching_tokens(token)
            if len(candidates) == 1:
                normalized_token = next(iter(candidates))
            else:
                raise KeyError("Unknown file token") from None

        if normalized_token in self._store:
            return self._store[normalized_token]
        if self.directory is None:
            raise KeyError("Unknown file token")

        record_path = self.directory / f"{normalized_token}.json"
        try:
            record = json.loads(record_path.read_text(encoding="utf-8"))
            path = Path(record["path"])
            if record["token"] != normalized_token or not _is_absolute_path(path):
                raise ValueError
        except (OSError, json.JSONDecodeError, KeyError, TypeError, ValueError):
            raise KeyError("Unknown file token") from None

        self._store[normalized_token] = path
        return path

    def _matching_tokens(self, prefix: str) -> set[str]:
        """Return registered full tokens uniquely matching a short prefix."""
        if not isinstance(prefix, str) or len(prefix.strip()) < 4:
            return set()
        needle = prefix.strip().lower()
        candidates: set[str] = set()
        for token in self._store:
            if token.lower().startswith(needle):
                candidates.add(token)
        if self.directory is not None:
            for record_path in self.directory.glob("*.json"):
                try:
                    token = str(uuid.UUID(record_path.stem))
                except ValueError:
                    continue
                if token.lower().startswith(needle):
                    candidates.add(token)
        return candidates


    def token_for_path(self, path: Path | str) -> Optional[str]:
        """Return the token already registered for this path, if any."""
        path_str = str(path)
        resolved = Path(path) if path_str.startswith("/") else Path(path).expanduser().resolve()
        for token, registered in self._store.items():
            if registered == resolved:
                return token
        if self.directory is not None:
            for record_path in self.directory.glob("*.json"):
                try:
                    token = str(uuid.UUID(record_path.stem))
                    record = json.loads(record_path.read_text(encoding="utf-8"))
                    rec_path = Path(record["path"])
                    if token not in self._store and rec_path == resolved:
                        self._store[token] = resolved
                        return token
                except (OSError, json.JSONDecodeError, KeyError, TypeError, ValueError):
                    continue
        return None

    def is_registered(self, token: str) -> bool:
        try:
            self.resolve(token)
        except KeyError:
            return False
        return True

    def revoke(self, token: str) -> None:
        """Remove a token from the registry (e.g. after staging is applied)."""
        try:
            normalized_token = str(uuid.UUID(token))
        except (TypeError, ValueError, AttributeError):
            return
        self._store.pop(normalized_token, None)
        if self.directory is not None:
            (self.directory / f"{normalized_token}.json").unlink(missing_ok=True)

    def list_tokens(self) -> list[dict]:
        """Return all registered tokens with their display names (no paths)."""
        entries = {
            token: {"token": token, "display_name": path.name}
            for token, path in self._store.items()
        }
        if self.directory is not None:
            for record_path in self.directory.glob("*.json"):
                try:
                    token = str(uuid.UUID(record_path.stem))
                    record = json.loads(record_path.read_text(encoding="utf-8"))
                    path = Path(record["path"])
                    if record["token"] == token and _is_absolute_path(path):
                        entries[token] = {"token": token, "display_name": path.name}
                except (OSError, json.JSONDecodeError, KeyError, TypeError, ValueError):
                    continue
        return list(entries.values())

    def clear(self) -> int:
        """Delete persisted registrations and return their count."""
        count = len(self.list_tokens())
        self._store.clear()
        if self.directory is not None:
            for record_path in self.directory.glob("*.json"):
                record_path.unlink(missing_ok=True)
        return count

    def _save(self, token: str, path: Path) -> None:
        """Atomically save one token mapping without exposing its path."""
        temporary_path = None
        try:
            self.directory.mkdir(parents=True, exist_ok=True)
            with tempfile.NamedTemporaryFile(
                "w",
                encoding="utf-8",
                dir=self.directory,
                delete=False,
                suffix=".tmp",
            ) as temporary:
                json.dump({"token": token, "path": str(path)}, temporary)
                temporary_path = Path(temporary.name)
            os.replace(temporary_path, self.directory / f"{token}.json")
        except OSError:
            if temporary_path is not None:
                try:
                    temporary_path.unlink(missing_ok=True)
                except OSError:
                    pass
            self._store.pop(token, None)
            raise RuntimeError(
                "Unable to save local registration; check the data directory "
                "and local permissions."
            ) from None

    def preview_tag_changes(file_path: str, new_tags: Dict[str, Any]) -> Dict[str, Any]:
    """Return a field-by-field diff without writing to the audio file."""
    current_tags = read_audio_metadata(file_path).get("tags", {})
    diff = {}
    invisible_fields = []
    for key in sorted(set(current_tags) | set(new_tags)):
        before = current_tags.get(key)
        after = new_tags.get(key, before)
        changed = (
            str(before) if before is not None else ""
        ) != (
            str(after) if after is not None else ""
        ) and key in new_tags
        diff[key] = {
            "before": before,
            "after": after if key in new_tags else before,
            "changed": changed,
            "invisible_chars_omitted": _has_invisible(before) or _has_invisible(after),
        }
        if _has_invisible(before) or _has_invisible(after):
            invisible_fields.append(key)
    return {
        "file_path": file_path,
        "has_changes": any(item["changed"] for item in diff.values()),
        "diff": diff,
        "invisible_chars_omitted": bool(invisible_fields),
        "fields_with_invisible_chars": invisible_fields,
    }

    new_tags: Dict[str, Any],

    """Write proposed tags after optional backup creation."""
    path = Path(file_path)
    if not path.is_file():
        raise FileNotFoundError(f"File not found: {file_path}")

    preview = preview_tag_changes(file_path, new_tags)
    if not preview["has_changes"]:
        return {
            "file_path": file_path,

        shutil.copy2(path, backup_path)

        write_mp4(audio, new_tags)

            if value is not None:
                audio[key] = [str(value)]
        audio.save()

        "file_path": file_path,
        "status": "success",
        "backup_created": str(backup_path) if backup_path else None,
        "fields_written": [key for key, change in preview["diff"].items()
                           if change["changed"]],
        "applied_diff": preview["diff"],
    }
